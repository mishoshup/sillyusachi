## Task 2: Rewrite scroll-snap.js with Proper Enterprise Lenis Config

<task type="auto">
  <name>Task 2: Enterprise Lenis setup with wrapper/content separation</name>
  <files>src/lib/scroll-snap.js</files>
  <action>
    REPLACE ENTIRE FILE with proper Lenis implementation.

    Key changes from current broken setup:
    - wrapper !== content via separate `.lenis-content` div (nested in Svelte template, NOT created via JS appendChild)
    - Single snap point per section (start only, no dual)
    - mandatory snap type for crisp snap (safe since no section overflows its snap)
    - Deferred RAF start after Svelte layout settles
    - ResizeObserver with rAF debounce for section height changes
    - Export cleanup + scrollToPage for nav

    ```javascript
    import Lenis from 'lenis';
    import Snap from 'lenis/snap';
    import 'lenis/dist/lenis.css';

    /**
     * Enterprise Lenis smooth scroll setup.
     * Content div (.lenis-content) MUST be manually nested in the Svelte template
     * to avoid Svelte 5 DOM reactivity issues.
     */
    export function setupLenis(wrapper, onPageChange) {
      const sections = Array.from(wrapper.querySelectorAll('[data-page]'));
      const content = wrapper.querySelector('.lenis-content');
      
      if (!sections.length || !content) {
        console.warn('[Lenis] Missing sections or .lenis-content');
        return { cleanup() {}, scrollToPage() {} };
      }

      // 1. Init Lenis with proper wrapper/content separation
      const lenis = new Lenis({
        wrapper,          // <main> element — the scrollport
        content,          // inner div — gets transform applied
        duration: 1.2,
        easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
        orientation: 'vertical',
        gestureOrientation: 'vertical',
        smoothWheel: true,
        syncTouch: true,
        autoResize: true,
      });

      // 2. Init Snap with single point per section
      const snap = new Snap(lenis, {
        type: 'mandatory',
        debounce: 100,
        lerp: 0.12,
        duration: 0.6,
      });

      // Single snap point per section
      sections.forEach((section) => {
        snap.addElement(section, { align: ['start'] });
      });

      // 3. Nav tracking via onSnapComplete
      // API: receives { index, value } — index is zero-based snap point index
      snap.onSnapComplete = ({ index }) => {
        const pageIdx = Math.min(index, sections.length - 1);
        if (pageIdx >= 0) onPageChange(pageIdx);
      };

      // 4. Deferred RAF (starts AFTER current frame for layout stability)
      let rafId;
      requestAnimationFrame(() => {
        function raf(time) {
          lenis.raf(time);
          rafId = requestAnimationFrame(raf);
        }
        rafId = requestAnimationFrame(raf);
      });

      // 5. ResizeObserver with rAF debounce for section height changes
      let resizePending = false;
      const resizeObserver = new ResizeObserver(() => {
        if (!resizePending) {
          resizePending = true;
          requestAnimationFrame(() => {
            snap.resize();
            resizePending = false;
          });
        }
      });
      sections.forEach((el) => resizeObserver.observe(el));

      // 6. Programmatic scroll for nav buttons
      function scrollToPage(index) {
        if (index >= 0 && index < sections.length) {
          lenis.scrollTo(sections[index], {
            offset: 0,
            immediate: false,
            lock: true,
            duration: 1.0,
          });
        }
      }

      // Cleanup
      return {
        cleanup() {
          cancelAnimationFrame(rafId);
          resizeObserver.disconnect();
          snap.destroy();
          lenis.destroy();
        },
        scrollToPage,
      };
    }
    ```

    IMPORTANT: Do NOT add while(appendChild) DOM relocation. Content div is already in the Svelte template.
  </action>
  <verify>ssh m4 "grep -cE 'wrapper:.*content:' src/lib/scroll-snap.js" — should return >=1 (shows wrapper !== content)</verify>

      return () => result.cleanup();

    function scrollToPage(index) {
      if (scrollToPageFn) scrollToPageFn(index);
    }
